# --- OpenAI ---
# OAI-SearchBot: retrieves pages to answer ChatGPT Search queries — allow
User-agent: OAI-SearchBot
Allow: /

# ChatGPT-User: retrieves a URL when a user explicitly requests it in ChatGPT — allow
# Note: OpenAI states user-initiated fetches "may not apply" robots.txt rules.
# This directive is included for completeness but cannot be guaranteed to bind.
User-agent: ChatGPT-User
Allow: /

# GPTBot: crawls pages to train OpenAI models — decline
User-agent: GPTBot

# --- Google ---
# Googlebot: Google web search index — allow
# Also controls AI Overviews and AI Mode (both use the Search index).

# and Gemini Apps / Vertex AI grounding — decline.
# Does NOT affect Google Search, AI Overviews, or AI Mode rankings.
User-agent: Google-Extended
Disallow: /

# --- Microsoft / Bing ---
# bingbot: Bing Search and Copilot grounding — allow.
# Microsoft does not publish a separate training-only crawler.
# Copilot cannot be decoupled from Bing: blocking bingbot removes both.

Allow: /
